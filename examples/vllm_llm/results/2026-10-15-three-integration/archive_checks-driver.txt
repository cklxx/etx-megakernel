def run(name,cmd,cwd=root,expected=0,env=None):
    start=time.monotonic();log=base/(name+'.log');stop=None
    print('STAGE',name,flush=True)
    with log.open('x') as out:
        p=subprocess.Popen(list(map(str,cmd)),cwd=cwd,env=env,stdout=out,stderr=subprocess.STDOUT,start_new_session=True)
        while p.poll() is None:
            if time.monotonic()-start>180 or shutil.disk_usage(base).free<4*1024**3:
                os.killpg(p.pid,signal.SIGTERM)
                try:p.wait(timeout=5)
                except subprocess.TimeoutExpired:os.killpg(p.pid,signal.SIGKILL);p.wait()
                stop='archive deadline/reserve';break
            print('PROGRESS',name,int(time.monotonic()-start),'seconds',flush=True)
            time.sleep(5)
    r=dict(name=name,command=list(map(str,cmd)),cwd=str(cwd),exit=p.returncode,seconds=time.monotonic()-start,expected_exit=expected,stop=stop)
    commands.append(r);(base/'archive-commands.json').write_text(json.dumps(commands,indent=2)+'\n')
    assert p.returncode==expected and stop is None,r
    print('DONE',name,p.returncode,flush=True)
